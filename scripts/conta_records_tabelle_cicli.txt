REM =============================================================================
REM Conteggio record delle tabelle esportate da export_cicli_tables.txt
REM (stessi filtri EXISTS: quanti record finirebbero nei CSV)
REM
REM Esecuzione (da cartella scripts/):
REM   sqlplus -s INTEX2/SD@192.168.1.10:1521/INT2DB @conta_records_tabelle_cicli.txt
REM
REM Oppure con output su file:
REM   sqlplus -s INTEX2/SD@192.168.1.10:1521/INT2DB @conta_records_tabelle_cicli.txt > conta_cicli_out.txt
REM =============================================================================

WHENEVER SQLERROR CONTINUE
WHENEVER OSERROR CONTINUE

SET ECHO OFF
SET FEEDBACK OFF
SET HEADING ON
SET PAGESIZE 100
SET LINESIZE 120
SET TRIMSPOOL ON
SET TRIMOUT ON
SET TERMOUT ON
SET VERIFY OFF
SET DEFINE OFF
SET TIMING ON
SET NUMWIDTH 12

COLUMN csv_file      FORMAT A28
COLUMN tabella       FORMAT A32
COLUMN n_export      FORMAT 999,999,999
COLUMN n_tabella     FORMAT 999,999,999

PROMPT === Conteggio record export cicli (filtri = export_cicli_tables.txt) ===

PROMPT --- Riepilogo: righe che finirebbero nei CSV ---

SELECT 'z09_cicli_testate.csv' AS csv_file,
       'Z09_TESTATE_CICLI_LAVORAZIONE' AS tabella,
       COUNT(*) AS n_export
  FROM INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE
UNION ALL
SELECT 'z08_cicli_fasi.csv',
       'Z08_CICLI_LAVORAZIONE',
       COUNT(*)
  FROM INTEX2.Z08_CICLI_LAVORAZIONE z8
 WHERE EXISTS (
   SELECT 1
   FROM INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE z9
   WHERE z9.Z09_CD_CLIENTE          = z8.Z08_CD_CLIENTE
     AND z9.Z09_CD_ARTICOLO_CLIENTE = z8.Z08_CD_ARTICOLO_CLIENTE
     AND z9.Z09_CD_CICLO            = z8.Z08_CD_CICLO
     AND z9.Z09_CD_LINEA            = z8.Z08_CD_LINEA
     AND z9.Z09_CD_STAGIONE         = z8.Z08_CD_STAGIONE
     AND z9.Z09_CD_COMPOSIZIONE     = z8.Z08_CD_COMPOSIZIONE
 )
UNION ALL
SELECT 'z02_fasi.csv',
       'Z02_FASI_DI_LAVORO',
       COUNT(*)
  FROM INTEX2.Z02_FASI_DI_LAVORO z2
 WHERE EXISTS (
   SELECT 1
   FROM INTEX2.Z08_CICLI_LAVORAZIONE z8
   WHERE z8.Z08_CD_FASE = z2.Z02_CD_FASE
     AND EXISTS (
       SELECT 1
       FROM INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE z9
       WHERE z9.Z09_CD_CLIENTE          = z8.Z08_CD_CLIENTE
         AND z9.Z09_CD_ARTICOLO_CLIENTE = z8.Z08_CD_ARTICOLO_CLIENTE
         AND z9.Z09_CD_CICLO            = z8.Z08_CD_CICLO
         AND z9.Z09_CD_LINEA            = z8.Z08_CD_LINEA
         AND z9.Z09_CD_STAGIONE         = z8.Z08_CD_STAGIONE
         AND z9.Z09_CD_COMPOSIZIONE     = z8.Z08_CD_COMPOSIZIONE
     )
 )
UNION ALL
SELECT 'r07_clienti.csv',
       'R07_CLIENTI',
       COUNT(*)
  FROM INTEX2.R07_CLIENTI r
 WHERE EXISTS (
   SELECT 1
   FROM INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE z9
   WHERE z9.Z09_CD_CLIENTE = r.R07_CD_CLIENTE
 )
UNION ALL
SELECT 'z11_stagioni.csv',
       'Z11_STAGIONI',
       COUNT(*)
  FROM INTEX2.Z11_STAGIONI z
 WHERE EXISTS (
   SELECT 1
   FROM INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE z9
   WHERE z9.Z09_CD_STAGIONE = z.Z11_CD_STAGIONE
 )
UNION ALL
SELECT 'cw4_composizioni.csv',
       'CW4_COMPOSIZIONI',
       COUNT(*)
  FROM INTEX2.CW4_COMPOSIZIONI c
 WHERE EXISTS (
   SELECT 1
   FROM INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE z9
   WHERE z9.Z09_CD_COMPOSIZIONE = c.CW4_CD_COMPOSIZIONE
 )
UNION ALL
SELECT 'cwa_articoli_cliente.csv',
       'CWA_ARTICOLI_CLIENTE',
       COUNT(*)
  FROM INTEX2.CWA_ARTICOLI_CLIENTE a
 WHERE EXISTS (
   SELECT 1
   FROM INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE z9
   WHERE z9.Z09_CD_CLIENTE          = a.CWA_CD_CLIENTE
     AND z9.Z09_CD_ARTICOLO_CLIENTE = a.CWA_CD_ARTICOLO_CLIENTE
 )
UNION ALL
SELECT 'cw0_linee.csv',
       'CW0_LINEA_TINTORIA',
       COUNT(*)
  FROM (
    SELECT DISTINCT l.CW0_CD_CLIENTE, l.CW0_CD_LINEA
    FROM INTEX2.CW0_LINEA_TINTORIA l
    WHERE EXISTS (
      SELECT 1
      FROM INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE z9
      WHERE z9.Z09_CD_CLIENTE = l.CW0_CD_CLIENTE
        AND z9.Z09_CD_LINEA   = l.CW0_CD_LINEA
    )
  );

           SELECT 1 FROM INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE z9

       (SELECT COUNT(*) FROM INTEX2.CW4_COMPOSIZIONI c
         WHERE EXISTS (
           SELECT 1 FROM INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE z9
           WHERE z9.Z09_CD_COMPOSIZIONE = c.CW4_CD_COMPOSIZIONE
         ))
  FROM DUAL
