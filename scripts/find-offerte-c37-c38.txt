PROMPT === 2d) Foreign key verso / da C37 e C38 ===
SELECT a.table_name, a.constraint_name, a.constraint_type, a.r_constraint_name,
       c.column_name, c.position
FROM all_constraints a
JOIN all_cons_columns c
  ON c.owner = a.owner AND c.constraint_name = a.constraint_name
WHERE a.owner = 'INTEX2'
  AND a.table_name IN ('C37_TESTATA_LISTINI','C38_RIGHE_LISTINI')
  AND a.constraint_type IN ('P','R','U')
ORDER BY a.table_name, a.constraint_type, a.constraint_name, c.position;

REM 5) Dove compaiono i codici univoci delle foto?
REM     Scansiona le colonne NUMBER di tabelle medie/piccole cercando
REM     39770 / 36625 / 40723. C38 e le tabelle enormi sono escluse apposta.

PROMPT === 5) Ricerca codici univoco 39770 / 36625 / 40723 ===
DECLARE
  v_found NUMBER;
BEGIN
  FOR c IN (
    SELECT table_name, column_name
    FROM all_tab_columns
    WHERE owner = 'INTEX2'
      AND data_type = 'NUMBER'
      AND table_name IN (
        'C37_TESTATA_LISTINI','C45_LISTINO_STAG_FASI_ARTF','C46_LISTINO_STAG_FASI_ARTC',
        'C52_PROTOTIPI_ATMAN','C40_LISTINO_MAT_STAGIONI','Z19_CICLI_STD_TESTATE',
        'Z09_TESTATE_CICLI_LAVORAZIONE'
      )
    ORDER BY table_name, column_id
  ) LOOP
    BEGIN
      EXECUTE IMMEDIATE
        'SELECT COUNT(*) FROM INTEX2.' || c.table_name ||
        ' WHERE ' || c.column_name || ' IN (39770, 36625, 40723) AND ROWNUM <= 1'
        INTO v_found;
      IF v_found > 0 THEN
        DBMS_OUTPUT.PUT_LINE('TROVATO: ' || c.table_name || '.' || c.column_name);
      END IF;
    EXCEPTION
      WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('ERRORE su ' || c.table_name || '.' || c.column_name || ': ' || SQLERRM);
    END;
  END LOOP;
  DBMS_OUTPUT.PUT_LINE('--- ricerca codici univoco terminata ---');
END;
/
