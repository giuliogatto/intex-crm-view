REM -----------------------------------------------------------------------------
REM 7) Join colonne candidate (catalogo, veloce) + join opzionale commentato
REM -----------------------------------------------------------------------------

PROMPT === 7) colonne ID/cliente/ciclo/stagione/prezzo ===
SELECT table_name, column_name, data_type
FROM all_tab_columns
WHERE owner IN ('INTEX2','INTEXGR','NEXSTYLE','INTEXGRIFFE')
  AND table_name IN (
    'C30_LISTINO_CLIENTI','C30_TEST_LIST_CLI','C33_LISTINO_CLIENTI_CICLI',
    'C52_PROTOTIPI_ATMAN','C52_DETT_LIST_CL',
    'Z09_TESTATE_CICLI_LAVORAZIONE','Z09_CICLI','Z08_CICLI_LAVORAZIONE'
  )
  AND (
       UPPER(column_name) LIKE '%ID%'
    OR UPPER(column_name) LIKE '%KEY%'
    OR UPPER(column_name) LIKE '%CLIENTE%'
    OR UPPER(column_name) LIKE '%STAGIONE%'
    OR UPPER(column_name) LIKE '%CICLO%'
    OR UPPER(column_name) LIKE '%UNIVOC%'
    OR UPPER(column_name) LIKE '%PREZZO%'
    OR UPPER(column_name) LIKE '%LIST%'
  )
ORDER BY owner, table_name, column_id;
