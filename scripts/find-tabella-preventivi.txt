-- 0c) Stessa tabella in altri schema? (dal catalogo risultava solo INTEX2)
SELECT owner, table_name
FROM all_tables
WHERE table_name LIKE '%LISTINO_CLIENTI%'
   OR table_name LIKE 'C30_%'
   OR table_name LIKE 'C52_%'
   OR table_name LIKE 'Z09_%'
ORDER BY owner, table_name;

-- 0d) Sinonimi (il gestionale potrebbe leggere un synonym, non la tabella locale)
SELECT owner, synonym_name, table_owner, table_name, db_link
FROM all_synonyms
WHERE synonym_name IN (
  'C30_LISTINO_CLIENTI','C31_LISTINO_CLIENTI_FASI','C33_LISTINO_CLIENTI_CICLI',
  'CX3_LISTINO_CLIENTI_CICLI','C52_PROTOTIPI_ATMAN','STORICO_DISPOSIZIONI',
  'Z08_CICLI_LAVORAZIONE','Z09_TESTATE_CICLI_LAVORAZIONE'
)
ORDER BY owner, synonym_name;

-- 0e) View con nome simile (dati "vivi" dietro a una view?)
SELECT owner, view_name
FROM all_views
WHERE view_name LIKE '%LISTINO_CLIENTI%'
   OR view_name LIKE '%OFFERT%'
   OR view_name LIKE '%PREVENT%'
   OR view_name LIKE 'C30%'
   OR view_name LIKE 'C52%'
ORDER BY owner, view_name;

-- 0f) Utente corrente e schema di default
SELECT USER AS utente_corrente, SYS_CONTEXT('USERENV','CURRENT_SCHEMA') AS current_schema FROM dual;

# 6) JOIN di verifica (solo se C30 e C33/Z09 hanno righe)
#     Attenzione: i nomi colonna FK (C30_ID, C33_ID_LISTINO, …) vanno
#     confermati dal DESC — se fallisce, correggi dopo la sezione struttura.
# =============================================================================

-- 6a) Colonne di join candidate (cercalesul catalogo)
SELECT table_name, column_name, data_type
FROM all_tab_columns
WHERE owner = 'INTEX2'
  AND table_name IN (
    'C30_LISTINO_CLIENTI','C33_LISTINO_CLIENTI_CICLI','CX3_LISTINO_CLIENTI_CICLI',
    'Z09_TESTATE_CICLI_LAVORAZIONE','C31_LISTINO_CLIENTI_FASI'
  )
  AND (
       UPPER(column_name) LIKE '%ID%'
    OR UPPER(column_name) LIKE '%KEY%'
    OR UPPER(column_name) LIKE '%CLIENTE%'
    OR UPPER(column_name) LIKE '%STAGIONE%'
    OR UPPER(column_name) LIKE '%CICLO%'
    OR UPPER(column_name) LIKE '%UNIVOC%'
    OR UPPER(column_name) LIKE '%PREZZO%'
  )
ORDER BY table_name, column_id;
