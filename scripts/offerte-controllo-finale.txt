PROMPT === A1) Chiavi primarie delle tabelle di decodifica (per confermare i JOIN) ===
SELECT c.table_name, c.constraint_name, cc.column_name, cc.position
FROM all_constraints c
JOIN all_cons_columns cc
  ON cc.owner = c.owner AND cc.constraint_name = c.constraint_name
WHERE c.owner = 'INTEX2'
  AND c.constraint_type = 'P'
  AND c.table_name IN (
    'R07_CLIENTI','Z11_STAGIONI','CW4_COMPOSIZIONI','CW0_LINEA_TINTORIA',
    'CWA_ARTICOLI_CLIENTE','Z09_TESTATE_CICLI_LAVORAZIONE','Z02_FASI_DI_LAVORO',
    'C50_CDC_TAB','C51_RIPARTIZIONI_COSTI_CDC'
  )
ORDER BY c.table_name, cc.position;

PROMPT
PROMPT === A2) Colonne descrittive delle tabelle di decodifica ===
SELECT table_name, column_name, data_type
FROM all_tab_columns
WHERE owner = 'INTEX2'
  AND table_name IN ('R07_CLIENTI','Z11_STAGIONI','CW4_COMPOSIZIONI','CW0_LINEA_TINTORIA',
                     'CWA_ARTICOLI_CLIENTE','Z02_FASI_DI_LAVORO')
  AND (column_name LIKE '%\_CD\_%' ESCAPE '\' OR column_name LIKE '%DS%' OR column_name LIKE '%RAGIONE%')
ORDER BY table_name, column_id;

PROMPT === A4) Significato dei flag: confronto 39770 (Prototipo) vs 36625 (Accettata) ===
SELECT t.C37_PRG            AS cod_univoco,
       t.C37_CD_CLIENTE     AS cliente,
       t.C37_CD_STAGIONE    AS stagione,
       t.C37_CD_CICLO       AS ciclo,
       t.C37_FLAG_PROTOTIPO AS fl_proto,
       t.C37_FLAG_VISIBILE  AS fl_vis,
       t.C37_ST_RECORD      AS st,
       z.Z09_FLAG_PROTOTIPO AS z09_fl_proto,
       z.Z09_NR_REVISIONE,
       z.Z09_PERC_FALLOSITA,
       t.C37_PZ_TOT_CAPO,
       t.C37_PZ_TOT_KG,
       t.C37_DATA_VALIDITA
FROM INTEX2.C37_TESTATA_LISTINI t
LEFT JOIN INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE z
  ON  z.Z09_CD_CLIENTE          = t.C37_CD_CLIENTE
  AND z.Z09_CD_ARTICOLO_CLIENTE = t.C37_CD_ARTICOLO
  AND z.Z09_CD_CICLO            = t.C37_CD_CICLO
WHERE t.C37_PRG IN (39770, 36625, 40723, 40067, 38594, 39636)
ORDER BY t.C37_PRG;

PROMPT === A5) Distribuzione flag su tutte le offerte ===
SELECT C37_FLAG_PROTOTIPO AS fl_proto, C37_FLAG_VISIBILE AS fl_vis, C37_ST_RECORD AS st, COUNT(*) AS n
FROM INTEX2.C37_TESTATA_LISTINI
GROUP BY C37_FLAG_PROTOTIPO, C37_FLAG_VISIBILE, C37_ST_RECORD
ORDER BY n DESC;

PROMPT
PROMPT === A6) Distribuzione Z09_FLAG_PROTOTIPO sulle offerte (join ciclo) ===
SELECT t.C37_FLAG_PROTOTIPO AS fl_proto, z.Z09_FLAG_PROTOTIPO AS z09_fl_proto, COUNT(*) AS n
FROM INTEX2.C37_TESTATA_LISTINI t
LEFT JOIN INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE z
  ON  z.Z09_CD_CLIENTE          = t.C37_CD_CLIENTE
  AND z.Z09_CD_ARTICOLO_CLIENTE = t.C37_CD_ARTICOLO
  AND z.Z09_CD_CICLO            = t.C37_CD_CICLO
GROUP BY t.C37_FLAG_PROTOTIPO, z.Z09_FLAG_PROTOTIPO

PROMPT === A10) Righe C38 dell'offerta 39770 (chiave a 7 campi) — confronto con la foto ===
SELECT r.C38_SEQUENZA, r.C38_CD_FORNITORE, r.C38_TP_PREZZO AS tp, r.C38_QTA_RIGA,
       r.C38_PZ_UNITARIO, r.C38_PZ_UNITARIO_KG, r.C38_CS_UNITARIO, r.C38_CS_UNITARIO_KG,
       r.C38_C37_PRG
FROM INTEX2.C37_TESTATA_LISTINI t
JOIN INTEX2.C38_RIGHE_LISTINI r
  ON  r.C38_CD_CLIENTE      = t.C37_CD_CLIENTE
  AND r.C38_CD_LINEA        = t.C37_CD_LINEA
  AND r.C38_CD_STAGIONE     = t.C37_CD_STAGIONE
  AND r.C38_CD_COMPOSIZIONE = t.C37_CD_COMPOSIZIONE
  AND r.C38_CD_ARTICOLO     = t.C37_CD_ARTICOLO
  AND r.C38_CD_CICLO        = t.C37_CD_CICLO
  AND r.C38_CD_VALUTA       = t.C37_CD_VALUTA
WHERE t.C37_PRG = 39770
ORDER BY r.C38_SEQUENZA;

SELECT * FROM (
  SELECT t.C37_PRG                AS cod_univoco,
         t.C37_CD_CLIENTE         AS cliente,
         r07.R07_RAGIONE_SOC      AS ragione_sociale,
         t.C37_CD_CICLO           AS ciclo,
         t.C37_CD_CICLO_CLI       AS ciclo_cli,
         z09.Z09_DS_CICLO         AS ds_ciclo,
         t.C37_CD_LINEA           AS linea,
         cw0.CW0_DS_LINEA         AS ds_linea,
         t.C37_CD_STAGIONE        AS stagione,
         z11.Z11_DS_STAGIONE      AS ds_stagione,
         t.C37_CD_COMPOSIZIONE    AS composizione,
         cw4.CW4_DS_COMPOSIZIONE  AS ds_composizione,
         t.C37_CD_ARTICOLO        AS articolo,
         cwa.CWA_DS_ARTICOLO_CLIENTE AS ds_articolo,
         t.C37_CD_VALUTA          AS valuta,
         t.C37_PZ_TOT_CAPO,
         t.C37_PZ_TOT_KG,
         t.C37_FLAG_PROTOTIPO     AS fl_proto,
         z09.Z09_NR_REVISIONE,
         z09.Z09_PERC_FALLOSITA,
         t.C37_DATA_VALIDITA
  FROM INTEX2.C37_TESTATA_LISTINI t
  LEFT JOIN INTEX2.R07_CLIENTI r07
    ON r07.R07_CD_CLIENTE = t.C37_CD_CLIENTE
  LEFT JOIN INTEX2.Z11_STAGIONI z11
    ON z11.Z11_CD_STAGIONE = t.C37_CD_STAGIONE
  LEFT JOIN INTEX2.CW4_COMPOSIZIONI cw4
    ON cw4.CW4_CD_COMPOSIZIONE = t.C37_CD_COMPOSIZIONE
  LEFT JOIN INTEX2.CW0_LINEA_TINTORIA cw0
    ON  cw0.CW0_CD_CLIENTE = t.C37_CD_CLIENTE
    AND cw0.CW0_CD_LINEA   = t.C37_CD_LINEA
  LEFT JOIN INTEX2.CWA_ARTICOLI_CLIENTE cwa
    ON  cwa.CWA_CD_CLIENTE  = t.C37_CD_CLIENTE
    AND cwa.CWA_CD_ARTICOLO = t.C37_CD_ARTICOLO
  LEFT JOIN INTEX2.Z09_TESTATE_CICLI_LAVORAZIONE z09
    ON  z09.Z09_CD_CLIENTE          = t.C37_CD_CLIENTE
    AND z09.Z09_CD_ARTICOLO_CLIENTE = t.C37_CD_ARTICOLO
    AND z09.Z09_CD_CICLO            = t.C37_CD_CICLO
  ORDER BY t.C37_PRG DESC
) WHERE ROWNUM <= 100;

PROMPT === B3) Righe (fasi/prezzi) delle ultime 20 offerte ===
SELECT t.C37_PRG AS cod_univoco,
       r.C38_SEQUENZA,
       r.C38_TP_PREZZO AS tp,
       r.C38_QTA_RIGA,
       r.C38_PZ_UNITARIO,
       r.C38_PZ_UNITARIO_KG,
       r.C38_CS_UNITARIO,
       r.C38_CS_UNITARIO_KG,
       r.C38_CD_FORNITORE
FROM (
  SELECT * FROM (
    SELECT * FROM INTEX2.C37_TESTATA_LISTINI ORDER BY C37_PRG DESC
  ) WHERE ROWNUM <= 20
) t
JOIN INTEX2.C38_RIGHE_LISTINI r
  ON  r.C38_CD_CLIENTE      = t.C37_CD_CLIENTE
  AND r.C38_CD_LINEA        = t.C37_CD_LINEA
  AND r.C38_CD_STAGIONE     = t.C37_CD_STAGIONE
  AND r.C38_CD_COMPOSIZIONE = t.C37_CD_COMPOSIZIONE
  AND r.C38_CD_ARTICOLO     = t.C37_CD_ARTICOLO
  AND r.C38_CD_CICLO        = t.C37_CD_CICLO
  AND r.C38_CD_VALUTA       = t.C37_CD_VALUTA
ORDER BY t.C37_PRG DESC, r.C38_SEQUENZA;
